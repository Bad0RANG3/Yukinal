cmake_minimum_required(VERSION 3.24)
project(yukinal_ssh_service VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  # The desktop installer must run on machines without a separately installed
  # Visual C++ Redistributable; keep the standalone service self-contained.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
endif()

option(YUKINAL_BUILD_TESTS "Build the SSH service tests" ON)
option(YUKINAL_FETCH_DEPS "Fetch pinned third-party source dependencies" ON)
option(YUKINAL_BUNDLE_WINDOWS_OPENSSL "Build a static OpenSSL crypto backend for Windows libssh2" ON)
option(YUKINAL_ENABLE_SSH_MOCK_TEST "Register the Paramiko SSH mock test when Paramiko is installed" ON)
option(YUKINAL_ENABLE_SSH_CONTAINER_TEST "Register the real OpenSSH container integration test" OFF)
option(YUKINAL_ENABLE_AGENT_TESTS "Register Python Agent Runtime tests when dependencies are installed" ON)
option(YUKINAL_BUILD_DESKTOP_HOST "Build the C++ desktop host that embeds the React UI" ON)

if(YUKINAL_FETCH_DEPS)
  include(FetchContent)

  if(WIN32 AND YUKINAL_BUNDLE_WINDOWS_OPENSSL)
    include(ExternalProject)
    find_program(YUKINAL_PERL_EXECUTABLE NAMES perl perl.exe REQUIRED)
    find_program(YUKINAL_NMAKE_EXECUTABLE NAMES nmake nmake.exe REQUIRED)
    execute_process(
      COMMAND "${YUKINAL_PERL_EXECUTABLE}" -MIPC::Cmd -MLocale::Maketext::Simple -e "1"
      RESULT_VARIABLE YUKINAL_PERL_MODULES_RESULT
      OUTPUT_QUIET
      ERROR_QUIET
    )
    if(NOT YUKINAL_PERL_MODULES_RESULT EQUAL 0)
      message(FATAL_ERROR
        "The Windows OpenSSL build needs a full Perl distribution (for example Strawberry Perl); "
        "the selected Perl is missing IPC::Cmd or Locale::Maketext::Simple. "
        "Set YUKINAL_PERL_EXECUTABLE to the full Perl executable.")
    endif()

    set(YUKINAL_OPENSSL_INSTALL_DIR "${CMAKE_BINARY_DIR}/_deps/openssl-install")
    set(YUKINAL_OPENSSL_CRYPTO_LIBRARY "${YUKINAL_OPENSSL_INSTALL_DIR}/lib/libcrypto.lib")
    set(YUKINAL_OPENSSL_SSL_LIBRARY "${YUKINAL_OPENSSL_INSTALL_DIR}/lib/libssl.lib")
    set(YUKINAL_OPENSSL_INCLUDE_DIR "${YUKINAL_OPENSSL_INSTALL_DIR}/include")
    file(MAKE_DIRECTORY "${YUKINAL_OPENSSL_INCLUDE_DIR}")
    set(YUKINAL_USE_BUNDLED_OPENSSL TRUE)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    set(OPENSSL_INCLUDE_DIR "${YUKINAL_OPENSSL_INCLUDE_DIR}")
    set(OPENSSL_CRYPTO_LIBRARY "${YUKINAL_OPENSSL_CRYPTO_LIBRARY}")
    set(OPENSSL_CRYPTO_LIBRARIES "${YUKINAL_OPENSSL_CRYPTO_LIBRARY}")
    set(OPENSSL_SSL_LIBRARY "${YUKINAL_OPENSSL_SSL_LIBRARY}")
    set(OPENSSL_SSL_LIBRARIES "${YUKINAL_OPENSSL_SSL_LIBRARY};${YUKINAL_OPENSSL_CRYPTO_LIBRARY}")
    set(OPENSSL_VERSION "3.5.9")
    set(OpenSSL_VERSION "3.5.9")
    list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(YUKINAL_OPENSSL_TARGET VC-WIN32)
    elseif(CMAKE_GENERATOR_PLATFORM MATCHES "ARM64" OR
           CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64" OR
           "$ENV{VSCMD_ARG_TGT_ARCH}" STREQUAL "arm64")
      set(YUKINAL_OPENSSL_TARGET VC-WIN64-ARM)
    else()
      set(YUKINAL_OPENSSL_TARGET VC-WIN64A)
    endif()

    ExternalProject_Add(yukinal_openssl
      URL "https://github.com/openssl/openssl/releases/download/openssl-3.5.9/openssl-3.5.9.tar.gz"
      URL_HASH SHA256=603f5602e2eef00d77fbd429d34dcd5822bb301757a1bc9cdb24c670f1eb859a
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE
      BUILD_IN_SOURCE TRUE
      CONFIGURE_COMMAND
        "${YUKINAL_PERL_EXECUTABLE}" <SOURCE_DIR>/Configure ${YUKINAL_OPENSSL_TARGET}
        no-shared no-tests no-apps no-docs no-makedepend no-asm
        --prefix=${YUKINAL_OPENSSL_INSTALL_DIR}
        --openssldir=${YUKINAL_OPENSSL_INSTALL_DIR}/ssl
      BUILD_COMMAND "${YUKINAL_NMAKE_EXECUTABLE}"
      INSTALL_COMMAND "${YUKINAL_NMAKE_EXECUTABLE}" install_sw
      USES_TERMINAL_CONFIGURE TRUE
      USES_TERMINAL_BUILD TRUE
      USES_TERMINAL_INSTALL TRUE
    )
  endif()

  set(JSON_BuildTests OFF CACHE INTERNAL "")
  FetchContent_Declare(
    nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
    GIT_SHALLOW TRUE
  )
  FetchContent_MakeAvailable(nlohmann_json)

  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(LIBSSH2_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(ENABLE_ZLIB_COMPRESSION OFF CACHE BOOL "" FORCE)
  if(WIN32 AND YUKINAL_BUNDLE_WINDOWS_OPENSSL)
    set(CRYPTO_BACKEND OpenSSL CACHE STRING "" FORCE)
    # Skip libssh2's optional BoringSSL probe: the bundled OpenSSL headers are
    # generated by the external build and do not exist during CMake configure.
    set(HAVE_BORINGSSL FALSE)
  elseif(WIN32)
    set(CRYPTO_BACKEND WinCNG CACHE STRING "" FORCE)
    set(ENABLE_ECDSA_WINCNG ON CACHE BOOL "" FORCE)
  else()
    # The service is shipped as a standalone desktop resource. Prefer static OpenSSL
    # archives so the installed service does not depend on a developer's OpenSSL path.
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    set(CRYPTO_BACKEND OpenSSL CACHE STRING "" FORCE)
  endif()
  FetchContent_Declare(
    libssh2
    GIT_REPOSITORY https://github.com/libssh2/libssh2.git
    # Includes the WinCNG fixed-width mpint normalization fix for intermittent KEX failures.
    GIT_TAG f216bbe3aa5ec02033de53ed0809a2461fc062c7
    GIT_SHALLOW FALSE
  )
  FetchContent_MakeAvailable(libssh2)
  if(WIN32 AND YUKINAL_BUNDLE_WINDOWS_OPENSSL AND TARGET libssh2_static)
    add_dependencies(libssh2_static yukinal_openssl)
  endif()

  FetchContent_Declare(
    cpp_httplib
    GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
    GIT_TAG v0.18.3
    GIT_SHALLOW TRUE
  )
  set(HTTPLIB_COMPILE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_INSTALL OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_TEST OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_WOLFSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_MBEDTLS_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(cpp_httplib)
else()
  find_package(nlohmann_json 3.11 CONFIG REQUIRED)
  find_package(libssh2 CONFIG REQUIRED)
  find_path(HTTPLIB_INCLUDE_DIR httplib.h REQUIRED)
endif()

find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto SSL)

add_library(yukinal_ssh_core
  src/config.cpp
  src/host_certificate.cpp
  src/host_key.cpp
  src/policy.cpp
  src/ssh.cpp
  src/http_service.cpp
)
target_include_directories(yukinal_ssh_core PUBLIC include)
if(YUKINAL_FETCH_DEPS)
  target_include_directories(yukinal_ssh_core PRIVATE ${cpp_httplib_SOURCE_DIR})
else()
  target_include_directories(yukinal_ssh_core PRIVATE ${HTTPLIB_INCLUDE_DIR})
endif()
target_link_libraries(yukinal_ssh_core
  PUBLIC nlohmann_json::nlohmann_json
  PRIVATE libssh2::libssh2 Threads::Threads OpenSSL::SSL OpenSSL::Crypto
)
target_compile_definitions(yukinal_ssh_core PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
if(WIN32 AND YUKINAL_FETCH_DEPS AND YUKINAL_BUNDLE_WINDOWS_OPENSSL)
  add_dependencies(yukinal_ssh_core yukinal_openssl)
endif()
if(WIN32)
  target_compile_definitions(yukinal_ssh_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(yukinal_ssh_core PRIVATE ws2_32 bcrypt)
endif()

add_executable(yukinal-ssh-service src/main.cpp)
target_link_libraries(yukinal-ssh-service PRIVATE yukinal_ssh_core)

if(YUKINAL_BUILD_TESTS)
  enable_testing()
  add_executable(yukinal_ssh_tests
    tests/test_main.cpp
    tests/test_config.cpp
    tests/test_host_key.cpp
    tests/test_policy.cpp
    tests/test_network.cpp
  )
  target_include_directories(yukinal_ssh_tests PRIVATE tests src)
  if(YUKINAL_FETCH_DEPS)
    target_include_directories(yukinal_ssh_tests PRIVATE ${cpp_httplib_SOURCE_DIR})
  else()
    target_include_directories(yukinal_ssh_tests PRIVATE ${HTTPLIB_INCLUDE_DIR})
  endif()
  target_compile_definitions(yukinal_ssh_tests PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
  target_link_libraries(yukinal_ssh_tests PRIVATE yukinal_ssh_core OpenSSL::SSL OpenSSL::Crypto)
  add_test(NAME yukinal_ssh_tests COMMAND yukinal_ssh_tests)

  find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
  add_test(
    NAME python_client_tests
    COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../agent/tests -p test_client.py -v
  )
  set_tests_properties(python_client_tests PROPERTIES
    ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/../agent/src"
  )

  add_test(
    NAME service_http_integration
    COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../agent/tests -p test_service_integration.py -v
  )
  set_tests_properties(service_http_integration PROPERTIES
    ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/../agent/src;YUKINAL_SSH_SERVICE_BINARY=$<TARGET_FILE:yukinal-ssh-service>"
    TIMEOUT 30
  )

  add_test(
    NAME ssh_container_harness_tests
    COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../scripts -p test_ssh_container.py -v
  )

  if(YUKINAL_ENABLE_AGENT_TESTS)
    execute_process(
      COMMAND ${Python3_EXECUTABLE} -c "import jsonschema, langgraph, mcp; import langgraph.checkpoint.sqlite.aio"
      RESULT_VARIABLE YUKINAL_AGENT_IMPORT_RESULT
      OUTPUT_QUIET
      ERROR_QUIET
    )
    if(YUKINAL_AGENT_IMPORT_RESULT EQUAL 0)
      add_test(
        NAME python_agent_tests
        COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../agent/tests -p test_agent_*.py -v
      )
      set_tests_properties(python_agent_tests PROPERTIES
        ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/../agent/src"
        TIMEOUT 90
      )
    else()
      message(STATUS "Agent dependencies are not installed; python_agent_tests is not registered (install agent[agent])")
    endif()
  endif()

  if(YUKINAL_ENABLE_SSH_MOCK_TEST)
    execute_process(
      COMMAND ${Python3_EXECUTABLE} -c "import paramiko"
      RESULT_VARIABLE YUKINAL_PARAMIKO_IMPORT_RESULT
      OUTPUT_QUIET
      ERROR_QUIET
    )
    if(YUKINAL_PARAMIKO_IMPORT_RESULT EQUAL 0)
      add_test(
        NAME ssh_mock_integration
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/test_ssh_mock.py
                --service-binary $<TARGET_FILE:yukinal-ssh-service>
      )
      set_tests_properties(ssh_mock_integration PROPERTIES
        ENVIRONMENT_MODIFICATION "PYTHONPATH=path_list_prepend:${CMAKE_CURRENT_SOURCE_DIR}/../agent/src"
        TIMEOUT 60
      )

      execute_process(
        COMMAND ${Python3_EXECUTABLE} -c "import jsonschema, langgraph, mcp; import langgraph.checkpoint.sqlite.aio"
        RESULT_VARIABLE YUKINAL_AGENT_MOCK_IMPORT_RESULT
        OUTPUT_QUIET
        ERROR_QUIET
      )
      if(YUKINAL_AGENT_MOCK_IMPORT_RESULT EQUAL 0)
        add_test(
          NAME agent_ssh_mock_integration
          COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/test_ssh_mock.py
                  --service-binary $<TARGET_FILE:yukinal-ssh-service> --agent-runtime-only
        )
        set_tests_properties(agent_ssh_mock_integration PROPERTIES
          ENVIRONMENT_MODIFICATION "PYTHONPATH=path_list_prepend:${CMAKE_CURRENT_SOURCE_DIR}/../agent/src"
          TIMEOUT 60
        )
      else()
        message(STATUS "Agent dependencies are not installed; agent_ssh_mock_integration is not registered")
      endif()
    else()
      message(STATUS "Paramiko is not installed; ssh_mock_integration is not registered (install agent[test])")
    endif()
  endif()

  if(YUKINAL_ENABLE_SSH_CONTAINER_TEST)
    add_test(
      NAME openssh_container_integration
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/test_ssh_container.py
              --service-binary $<TARGET_FILE:yukinal-ssh-service>
    )
    set_tests_properties(openssh_container_integration PROPERTIES
      ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/../agent/src"
      TIMEOUT 180
    )
  endif()
endif()

if(YUKINAL_BUILD_DESKTOP_HOST)
  if(YUKINAL_FETCH_DEPS)
    set(WEBVIEW_BUILD ON CACHE BOOL "" FORCE)
    set(WEBVIEW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_ENABLE_PACKAGING OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(
      webview
      GIT_REPOSITORY https://github.com/webview/webview.git
      GIT_TAG 3ab4b5d722438fc8a13e6ca830c5e2372d19a01d
      GIT_SHALLOW FALSE
    )
    FetchContent_MakeAvailable(webview)
  else()
    find_package(webview 0.12 CONFIG REQUIRED)
  endif()

  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../package.json" YUKINAL_ROOT_PACKAGE_JSON)
  string(REGEX MATCH "\"version\"[ \t]*:[ \t]*\"([^\"]+)\"" YUKINAL_VERSION_MATCH
    "${YUKINAL_ROOT_PACKAGE_JSON}")
  if(NOT YUKINAL_VERSION_MATCH)
    message(FATAL_ERROR "Could not read the desktop version from the root package.json")
  endif()
  set(YUKINAL_DESKTOP_VERSION "${CMAKE_MATCH_1}")

  add_library(yukinal_desktop_host_core
    desktop_host/src/ipc_router.cpp
  )
  target_include_directories(yukinal_desktop_host_core PUBLIC desktop_host/include)
  target_link_libraries(yukinal_desktop_host_core PUBLIC nlohmann_json::nlohmann_json)
  target_link_libraries(yukinal_desktop_host_core PRIVATE Threads::Threads)
  target_compile_definitions(yukinal_desktop_host_core PUBLIC
    YUKINAL_DESKTOP_VERSION="${YUKINAL_DESKTOP_VERSION}"
  )
  if(WIN32)
    target_compile_definitions(yukinal_desktop_host_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(yukinal_desktop_host_core PRIVATE shell32)
    add_executable(yukinal-desktop-host WIN32 desktop_host/src/main.cpp)
    target_compile_definitions(yukinal-desktop-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    add_executable(yukinal-desktop-host desktop_host/src/main.cpp)
  endif()
  target_link_libraries(yukinal-desktop-host PRIVATE yukinal_desktop_host_core webview::core)
  if(YUKINAL_FETCH_DEPS)
    target_link_libraries(yukinal-desktop-host PRIVATE httplib::httplib)
  else()
    target_include_directories(yukinal-desktop-host PRIVATE ${HTTPLIB_INCLUDE_DIR})
  endif()

  if(YUKINAL_BUILD_TESTS)
    add_executable(yukinal_desktop_host_tests
      desktop_host/tests/test_ipc_router.cpp
    )
    target_link_libraries(yukinal_desktop_host_tests PRIVATE yukinal_desktop_host_core)
    add_test(NAME desktop_host_tests COMMAND yukinal_desktop_host_tests)
    add_test(NAME desktop_host_self_check COMMAND yukinal-desktop-host --self-check)
  endif()
endif()
